const getBriefingText = () => {
  const parts = [];
  
  // Pending messages part
  if (messages.length > 0) {
    parts.push({
      text: `You have `,
      highlight: `${messages.length} messages ready to send`,
      after: ``
    });
  }

  // High churn risk part — replaces generic drifting count
  if (highRiskDrifting.length > 0 && mostUrgent) {
    const firstName = mostUrgent.name.split(' ')[0];
    const lastName = mostUrgent.name.split(' ').slice(-1)[0];
    const displayName = `${firstName} ${lastName[0]}.`;
    const othersCount = highRiskDrifting.length - 1;
    
    const daysAway = mostUrgent.last_visit_date 
      ? Math.floor((Date.now() - new Date(
          mostUrgent.last_visit_date).getTime()) / 86400000)
      : 0;
    
    const timeAway = daysAway < 30 ? 'a few weeks' 
      : daysAway < 60 ? 'over a month'
      : daysAway < 120 ? 'a couple of months'
      : 'several months';

    const othersText = othersCount > 0 
      ? ` and ${othersCount} other${othersCount > 1 ? 's' : ''}` 
      : '';

    return buildBriefing(
      messages.length,
      highRiskDrifting.length,
      displayName,
      othersText,
      timeAway,
      highRiskPlayers.length
    );
  }

  // Fallback — no high risk players, use old logic
  const atRisk = players.filter(p =>
    getEngagementState(p.last_visit_date) === 'lapsing' ||
    getEngagementState(p.last_visit_date) === 'dormant'
  );
  
  if (atRisk.length === 0 && messages.length === 0) {
    return { 
      text: "Everything looks good — no players need attention right now.",
      highlights: []
    };
  }

  return buildFallbackBriefing(messages.length, atRisk.length);
};

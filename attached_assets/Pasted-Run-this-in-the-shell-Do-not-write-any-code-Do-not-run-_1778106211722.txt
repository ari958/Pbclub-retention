const API_KEY = process.env.PODPLAY_API_KEY;
const KERRIE_ID = 'isapXoITEeOC5yAotTi9OzITTDn1';
const START = '2025-01-01T00:00:00.000Z';
const END = new Date().toISOString();
const sleep = ms => new Promise(r => setTimeout(r, ms));

async function run() {
  let page = 1;
  let totalEvents = 0;
  let kerrieEvents = [];
  let done = false;

  while (!done) {
    const url = PODPLAY_BASE + '/events' +
      '?startTime=' + encodeURIComponent(START) +
      '&endTime=' + encodeURIComponent(END) +
      '&page=' + page + '&ipp=100';

    const res = await fetch(url, {
      headers: { 'x-api-key': API_KEY }
    });
    const j = await res.json();
    const items = j.items || [];
    totalEvents += items.length;

    const now = new Date();
    const pastEvents = items.filter(e =>
      !e.isCanceled &&
      new Date(e.startTime) <= now &&
      !['COURT_RESERVATION','CO_WORKING','INTERNAL',
        'PRIVATE','PRIVATE_EVENT','OPERATIONS',
        'CONTRACT_TIME'].includes((e.subtype||'').toUpperCase())
    );

      );
